// ================================================================
// PART A: STACK
// ================================================================

class Stack {
  constructor() {
    // Create the internal storage.
    this.items = [];
  }

  // Add an element to the top of the stack.
  push(element) {
    this.items.push(element);
  }

  // Remove and return the top element.
  pop() {
    return this.items.pop();
  }

  // Return the top element without removing it.
  peek() {
    return this.items[this.items.length - 1];
  }

  // Check if the stack is empty.
  isEmpty() {
    return this.items.length === 0;
  }

  // Return the number of elements.
  size() {
    return this.items.length;
  }

  // Return all elements as text.
  toString() {
    return this.items.join(', ');
  }
}

console.log('\nPART A: UNDO HISTORY');

const actionHistory = new Stack();

// Push the three required actions.
actionHistory.push('Opened Ticket #101');
actionHistory.push('Assigned Ticket #101');
actionHistory.push('Closed Ticket #101');

// Display the required stack information.
console.log('Top Action:', actionHistory.peek());

const undoAction = actionHistory.pop();
console.log('Undo Action:', undoAction);

console.log('Remaining Actions:', actionHistory.toString());
console.log('Stack Size:', actionHistory.size());

// PART B: QUEUE
// ================================================================

class Queue {
  constructor() {
    // Create the internal storage.
    this.items = [];
  }

  // Add an element at the back of the queue.
  enqueue(element) {
    this.items.push(element);
  }

  // Remove and return the front element.
  dequeue() {
    return this.items.shift();
  }

  // Return the front element without removing it.
  front() {
    return this.items[0];
  }

  // Check if the queue is empty.
  isEmpty() {
    return this.items.length === 0;
  }

  // Return the number of elements.
  size() {
    return this.items.length;
  }

  // Return all elements as text.
  toString() {
    return this.items.join(', ');
  }
}

// ================================================================

class PriorityQueue {
  constructor() {
    // Create the internal storage.
    this.items = [];
  }

  // Add a request according to its priority.
  enqueue(request, priority) {
    const item = {
      request: request,
      priority: priority
    };

    let inserted = false;

    // Find the correct insertion position.
    // Using > preserves arrival order for equal priorities.
    for (let i = 0; i < this.items.length; i++) {
      if (priority < this.items[i].priority) {
        this.items.splice(i, 0, item);
        inserted = true;
        break;
      }
    }

    // Add to the end if no higher-priority position was found.
    if (!inserted) {
      this.items.push(item);
    }
  }

  // Remove and return the first item.
  dequeue() {
    return this.items.shift();
  }

  // Return the number of items.
  size() {
    return this.items.length;
  }

  // Display every request and priority.
  print() {
    this.items.forEach(item => {
      console.log(
        item.request + ', Priority ' + item.priority
      );
    });
  }
}

class Node {
  constructor(element) {
    // Store the element.
    this.element = element;

    // Set the next node to null.
    this.next = null;
  }
}

class LinkedList {
  constructor() {
    // Set the length to zero.
    this.length = 0;

    // Set the head to null.
    this.head = null;
  }

  // Add a node at the end.
  append(element) {
    const node = new Node(element);

    if (this.head === null) {
      this.head = node;
    } else {
      let current = this.head;

      while (current.next !== null) {
        current = current.next;
      }

      current.next = node;
    }

    this.length++;
  }

  // Insert an element at a specific position.
  insert(position, element) {
    if (position < 0 || position > this.length) {
      return false;
    }

    const node = new Node(element);

    // Insert at the beginning.
    if (position === 0) {
      node.next = this.head;
      this.head = node;
    } else {
      let previous = this.head;

      for (let i = 1; i < position; i++) {
        previous = previous.next;
      }

      node.next = previous.next;
      previous.next = node;
    }

    this.length++;
    return true;
  }

  // Remove a node at a specific position.
  removeAt(position) {
    if (position < 0 || position >= this.length) {
      return null;
    }

    let current = this.head;

    // Remove the first node.
    if (position === 0) {
      this.head = current.next;
    } else {
      let previous = this.head;

      for (let i = 1; i < position; i++) {
        previous = previous.next;
      }

      current = previous.next;
      previous.next = current.next;
    }

    this.length--;
    return current.element;
  }

  // Find the position of an element.
  indexOf(element) {
    let current = this.head;
    let index = 0;

    while (current !== null) {
      if (current.element === element) {
        return index;
      }

      current = current.next;
      index++;
    }

    return -1;
  }

  // Remove an element by its value.
  remove(element) {
    const index = this.indexOf(element);

    if (index !== -1) {
      return this.removeAt(index);
    }

    return null;
  }

  // Check if the linked list is empty.
  isEmpty() {
    return this.length === 0;
  }

  // Return the number of nodes.
  size() {
    return this.length;
  }

  // Traverse and return all elements as text.
  toString() {
    const elements = [];
    let current = this.head;

    while (current !== null) {
      elements.push(current.element);
      current = current.next;
    }

    return elements.join(', ');
  }
}
