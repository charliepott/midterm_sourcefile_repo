'use strict';

console.log('STUDENT SCORE ARRAY ANALYZER VERSION 3');
console.log('');

const studentNames = [
    'Ana',
    'Ben',
    'Carla',
    'Daniel',
    'Ella'
];

const studentScores = [85, 72, 91, 64, 78];

const activityScores = [
    [80, 85, 90],
    [70, 72, 74],
    [88, 92, 95],
    [60, 65, 67],
    [75, 78, 81]
];

console.log('PART A: ORIGINAL RECORDS');

for (let i = 0; i < studentNames.length; i++) {
    console.log(i + ' - ' + studentNames[i] + ': ' + studentScores[i]);
}

console.log('');

console.log('PART B: ADD NEW STUDENTS');

studentNames.push('Faith');
studentScores.push(88);

studentNames.unshift('Aaron');
studentScores.unshift(76);

for (let i = 0; i < studentNames.length; i++) {
    console.log(i + ' - ' + studentNames[i] + ': ' + studentScores[i]);
}

console.log('Number of students: ' + studentNames.length);
console.log('');

console.log('PART C: REMOVE AND REPLACE RECORDS');

studentNames.shift();
studentScores.shift();

studentNames.pop();
studentScores.pop();

let danielIndex = studentNames.indexOf('Daniel');

if (danielIndex !== -1) {
    studentNames.splice(danielIndex, 1);
    studentScores.splice(danielIndex, 1);
}

studentNames.splice(2, 0, 'Grace');
studentScores.splice(2, 0, 82);

console.log('Student Names:', studentNames);
console.log('Student Scores:', studentScores);

console.log('');

console.log('PART D: CLASS STATISTICS');

let totalScore = 0;

for (let i = 0; i < studentScores.length; i++) {
    totalScore += studentScores[i];
}

let averageScore = totalScore / studentScores.length;

let highestScore = studentScores[0];
let lowestScore = studentScores[0];

for (let i = 1; i < studentScores.length; i++) {
    if (studentScores[i] > highestScore) {
        highestScore = studentScores[i];
    }

    if (studentScores[i] < lowestScore) {
        lowestScore = studentScores[i];
    }
}

console.log('Total Score: ' + totalScore);
console.log('Average Score: ' + averageScore.toFixed(2));
console.log('Highest Score: ' + highestScore);
console.log('Lowest Score: ' + lowestScore);

console.log('');

console.log('PART E: PASSING SCORE ANALYSIS');

const passingScore = 75;

const passingScores = studentScores.filter(score => score >= passingScore);

const results = studentScores.map(score =>
    score >= passingScore ? 'Passed' : 'Failed'
);

const hasFailed = studentScores.some(score => score < passingScore);

const allPassed = studentScores.every(score => score >= passingScore);

console.log('Passing Scores:', passingScores);
console.log('Passed or Failed:', results);
console.log('At least one student failed: ' + hasFailed);
console.log('All students passed: ' + allPassed);

console.log('');

console.log('PART F: SORT AND SEARCH');

const sortedScores = Array.from(studentScores);

sortedScores.sort((a, b) => b - a);

console.log('Sorted Scores:', sortedScores);
console.log('Contains score 91: ' + studentScores.includes(91));
console.log('Index of score 91: ' + studentScores.indexOf(91));

console.log('');

console.log('PART G: ACTIVITY SCORES');

const activityAverages = [];

for (let i = 0; i < activityScores.length; i++) {
    let rowTotal = 0;

    console.log(studentNames[i] + "'s activity scores:");

    for (let j = 0; j < activityScores[i].length; j++) {
        console.log(activityScores[i][j]);
        rowTotal += activityScores[i][j];
    }

    let rowAverage = rowTotal / activityScores[i].length;

    activityAverages.push(rowAverage);

    console.log(
        studentNames[i] +
        "'s activity average: " +
        rowAverage.toFixed(2)
    );

    console.log('');
}

console.log('Activity Averages:', activityAverages);

console.log('');

console.log('PART H: TEXT SUMMARY');

console.log('Students: ' + studentNames.join(' | '));
